\begin{equation}
\quad K_y = K + \sigma_n^2 I_N
\end{equation}

\begin{equation}
\log p(\mathbf{y} \mid X, \boldsymbol{\theta}) = -\frac{1}{2} \mathbf{y}^T K_y^{-1} \mathbf{y} - \frac{1}{2} \log |K_y| - \frac{N}{2} \log(2\pi)

\begin{equation}
\frac{\partial \log p(\mathbf{y} \mid X, \boldsymbol{\theta})}{\partial \theta_j} = \frac{1}{2} \operatorname{tr}\left( \left(\boldsymbol{\alpha}\boldsymbol{\alpha}^T - K_y^{-1}\right) \frac{\partial K_y}{\partial \theta_j} \right), \quad \boldsymbol{\alpha} = K_y^{-1}\mathbf{y}
\end{equation}

\section{Leave-One-Out Cross-Validation}
\label{sec:loo_nlpl}
\noindent \textit{\citep[Bab~5.3 \& Bab~5.4.2, hlm.~111 \& hlm.~116--118, Persamaan~(5.6) \& (5.11) ]{rasmussen2006gaussian}}

\begin{equation}
L_{\text{LOO}}(\boldsymbol{\theta}) = -\sum_{i=1}^N \log p(y_i \mid X, \mathbf{y}_{-i}, \boldsymbol{\theta}) = \frac{1}{2} \sum_{i=1}^N \left( -\log [K_y^{-1}]_{ii} + \frac{([K_y^{-1}\mathbf{y}]_i)^2}{[K_y^{-1}]_{ii}} + \log(2\pi) \right)
\end{equation}

\begin{align}
\frac{\partial L_{\text{LOO}}(\boldsymbol{\theta})}{\partial \theta_j} &= \sum_{i=1}^{n} \left( \alpha_i [Z_j \boldsymbol{\alpha}]_i - \frac{1}{2} \left( 1 + \frac{\alpha_i^2}{[K^{-1}]_{ii}} \right) [Z_j K^{-1}]_{ii} \right) / [K^{-1}]_{ii}.
\end{align}
