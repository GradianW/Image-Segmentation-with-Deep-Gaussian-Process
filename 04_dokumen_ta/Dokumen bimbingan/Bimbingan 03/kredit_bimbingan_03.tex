\section{Mencari Sample dari Distribusi Gaussian Process}
\label{sec:mekanisme_sampling}

Sampel fungsi berkorelasi dapat dibangkitkan dari \textit{random noise} independen berkat sifat distribusi Gaussian multivariat yang tetap berdistribusi Gaussian saat dikenai \textit{affine transformation}:

\begin{teorema}[Transformasi Affine Vektor Acak Gaussian \citep{hogg2013introduction}]
\label{thm:affine_trans}
Jika $\mathbf{u} \sim \mathcal{N}(\mathbf{\mu}_u, \Sigma_u)$ adalah vektor acak Gaussian berdimensi $n$, dan diberikan matriks transformasi konstan $A \in \mathbb{R}^{m \times n}$ serta vektor translasi konstan $\mathbf{b} \in \mathbb{R}^m$, maka vektor hasil transformasi affine $\mathbf{f} = A \mathbf{u} + \mathbf{b}$ terdistribusi secara Gaussian multivariat berdimensi $m$:
\begin{equation}
\mathbf{f} = A \mathbf{u} + \mathbf{b} \sim \mathcal{N}\left(A \mathbf{\mu}_u + \mathbf{b}, \; A \Sigma_u A^T\right)
\end{equation}
\end{teorema}

Karena komputer hanya mampu membangkitkan \textit{noise} independen $\mathbf{u} \sim \mathcal{N}(\mathbf{0}, I_{N_*})$, kita memerlukan transformasi affine untuk membentuk korelasi antar-titik. Berdasarkan Teorema~\ref{thm:affine_trans}, kita mengalikan \textit{noise} tersebut dengan faktor Cholesky $L$ (di mana $\tilde{K} = L L^T$) dan menambahkan nilai rata-rata $\mathbf{m}_*$. Vektor sampel $\mathbf{f}_* = \mathbf{m}_* + L\mathbf{u}$ yang terbentuk secara eksak memenuhi:
\begin{align}
\mathbb{E}[\mathbf{f}_*] &= \mathbf{m}_* + L \mathbb{E}[\mathbf{u}] = \mathbf{m}_* + L(\mathbf{0}) = \mathbf{m}_* \\
\operatorname{cov}(\mathbf{f}_*) &= \mathbb{E}\left[(L\mathbf{u})(L\mathbf{u})^T\right] = L \, \mathbb{E}[\mathbf{u}\mathbf{u}^T] L^T = L I_{N_*} L^T = L L^T = \tilde{K} \approx K(X_*, X_*)
\end{align}
sehingga $\mathbf{f}_* \sim \mathcal{N}(\mathbf{m}_*, K)$.

\subsection{Algoritma Komputasi Sampling GP}

Sampling GP diimplementasikan melalui tahapan berikut:
\begin{enumerate}
    \item \textbf{Evaluasi \& Regularisasi Matriks Kovariansi}: Hitung matriks $K = K(X_*, X_*)$ dan tambahkan nilai \textit{jitter} diagonal $\epsilon \approx 10^{-6}$ untuk menjamin sifat definit positif terhadap galat numerik:
    \begin{equation}
    \tilde{K} = K + \epsilon I_{N_*}
    \end{equation}
    \item \textbf{Faktorisasi Cholesky}: Dekomposisi $\tilde{K}$ menjadi matriks segitiga bawah unik $L \in \mathbb{R}^{N_* \times N_*}$:
    \begin{equation}
    \tilde{K} = L L^T
    \end{equation}
    \item \textbf{Generating Random Noise}: Bangkitkan vektor \textit{noise} baku $\mathbf{u} \sim \mathcal{N}(\mathbf{0}, I_{N_*})$. Untuk $S$ kurva sekaligus, gunakan matriks $U \in \mathbb{R}^{N_* \times S}$ dengan $U_{ij} \sim \mathcal{N}(0, 1)$.
    \item \textbf{Transformasi Affine}: Hitung sampel fungsi melalui:
    \begin{equation}
    \mathbf{f}_* = \mathbf{m}_* + L \mathbf{u} \quad \left(\text{atau } F = \mathbf{m}_* \mathbf{1}_S^T + L U \text{ untuk } S \text{ sampel}\right)
    \end{equation}
\end{enumerate}

\subsection{Sampling Prior versus Posterior}

Mekanisme transformasi Cholesky berlaku serupa baik sebelum maupun sesudah mengamati data observasi $(X, \mathbf{y})$:
\begin{itemize}
    \item \textbf{Prior Sampling}: Menggunakan mean $\mathbf{m}_*$ dan matriks kovariansi $K(X_*, X_*)$:
    \begin{equation}
    \mathbf{f}_* \sim \mathcal{N}\big(\mathbf{\mathbf{m}_*}, K(X_*, X_*)\big) \implies \mathbf{f}_* = \mathbf{m}_* + L \mathbf{u}, \quad \text{dengan } L = \operatorname{cholesky}\big(K(X_*, X_*) + \epsilon I_{N_*}\big)
    \end{equation}
    \item \textbf{Posterior Sampling}: Menggunakan distribusi posterior Gaussian Process Regression \citep{rasmussen2006gaussian}:
    \begin{equation}
    \mathbf{f}_* \mid X, \mathbf{y}, X_* \sim \mathcal{N}\big( \bar{\mathbf{f}}_*, \operatorname{cov}(\mathbf{f}_*) \big)
    \end{equation}
    dengan parameter analitik:
    \begin{align}
    \bar{\mathbf{f}}_* &= K(X_*, X) \left[ K(X, X) + \sigma_n^2 I_N \right]^{-1} \mathbf{y} \\
    \operatorname{cov}(\mathbf{f}_*) &= K(X_*, X_*) - K(X_*, X) \left[ K(X, X) + \sigma_n^2 I_N \right]^{-1} K(X, X_*)
    \end{align}
    Sampel fungsi posterior dihitung melalui:
    \begin{equation}
    \mathbf{f}_{*,\text{post}} = \bar{\mathbf{f}}_* + L_{\text{post}} \mathbf{u}, \quad \text{dengan } L_{\text{post}} = \operatorname{cholesky}\big(\operatorname{cov}(\mathbf{f}_*) + \epsilon I_{N_*}\big)
    \end{equation}
\end{itemize}

\subsection{Operasi Aljabar Kernel}
\label{subsec:operasi_kernel}

Kombinasi kernel dasar menghasilkan fungsi kovariansi baru yang tetap valid dan definit positif \citep{rasmussen2006gaussian}:
\begin{itemize}
    \item \textbf{Penjumlahan}: $k(\mathbf{x}, \mathbf{x}') = k_1(\mathbf{x}, \mathbf{x}') + k_2(\mathbf{x}, \mathbf{x}')$ \\
    Merepresentasikan superposisi proses independen $f(\mathbf{x}) = f_1(\mathbf{x}) + f_2(\mathbf{x})$.
    
    \item \textbf{Perkalian}: $k(\mathbf{x}, \mathbf{x}') = k_1(\mathbf{x}, \mathbf{x}') \cdot k_2(\mathbf{x}, \mathbf{x}')$ \\
    Merepresentasikan interaksi atau modulasi proses $f(\mathbf{x}) = f_1(\mathbf{x}) \cdot f_2(\mathbf{x})$.
\end{itemize}
