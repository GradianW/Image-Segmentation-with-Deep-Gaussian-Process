\begin{abstract}
\noindent Dokumen ringkas ini merangkum esensi komputasi dan pemodelan \textit{Deep Gaussian Process} (DGP). Pembahasan difokuskan pada tiga pilar utama: (1) interpretasi konsep ``banyak neuron'' pada DGP (teorema Neal lebar tak terhingga vs dimensi representasi laten terhitung), (2) diagram alur arsitektur propagasi hierarkis dua lapisan ($L = 2$) dengan trik reparameterisasi, serta (3) simulasi perhitungan matriks manual menggunakan angka \textit{dummy} konkret ($N = 3$) pada titik interpolasi baru ($x_* = 3.0$). Ditunjukkan secara analitik bagaimana ketidakpastian lapisan tersembunyi merambat melalui \textit{Law of Total Variance} menuju luaran akhir.
\end{abstract}

\section{Konsep ``Banyak Neuron'' pada Deep Gaussian Process}
\label{sec:konsep_neuron}

\noindent Dalam \textit{Deep Neural Network} (DNN), neuron adalah unit komputasi skalar berbobot: $h_j = \sigma(\sum_i W_{ji} x_i + b_j)$. Ketika beralih ke \textit{Deep Gaussian Process} (DGP), konsep ``neuron'' dipahami melalui \textbf{dua sudut pandang}:

\begin{enumerate}
    \item \textbf{Sudut Pandang Teoretis (Teorema Neal, 1995) $\to$ Neuron Tak Terhingga ($\infty$)}: \\
    Jika bobot dan bias neural network berprior Gaussian independen $W_{ji} \sim \mathcal{N}(0, \sigma_w^2)$ diekspansi lebarnya menuju tak berhingga ($M \to \infty$), distribusi fungsi jaringan tersebut berkonvergensi eksak ke proses Gaussian:
    \begin{equation}
        \lim_{M \to \infty} f_{\mathrm{NN}}(\mathbf{x}) \equiv \mathcal{GP}\left( 0, \; k(\mathbf{x}, \mathbf{x}') \right)
    \end{equation}
    Artinya, \textbf{setiap satu fungsi GP secara inheren memuat kapasitas representasi $\infty$ neuron berbobot acak} di ruang bobot (\textit{weight space}).
    
    \item \textbf{Sudut Pandang Arsitektural Praktis $\to$ Dimensi Laten Terhitung ($D_\ell$)}: \\
    Yang diatur oleh arsitek DGP pada setiap lapisan tersembunyi adalah \textbf{dimensi representasi laten} $D_\ell$ (jumlah kanal laten). Jika $D_\ell = 1$, terdapat 1 fungsi laten $h \in \mathbb{R}$. Jika $D_\ell = 2$ atau $D_\ell = 16$, terdapat beberapa kanal laten independen:
    \begin{equation}
        \mathbf{h}^{(\ell)} = \begin{bmatrix} h_1^{(\ell)} \\ \vdots \\ h_{D_\ell}^{(\ell)} \end{bmatrix}, \quad h_d^{(\ell)} \sim \mathcal{GP}\left(0, \; k_d^{(\ell)}\right), \quad d = 1, \dots, D_\ell
    \end{equation}
    Keunggulan \textit{kernel trick}: berapapun jumlah kanal laten $D_\ell$, \textbf{dimensi matriks kovariansi $K^{(\ell)}$ tetap berukuran $[N \times N]$} (hanya bergantung pada jumlah data $N$, bukan jumlah dimensi laten).
\end{enumerate}

\subsection{Tantangan Non-linearitas dan Reparameterization Trick}
\label{subsec:reparam}
Karena luaran Lapisan 1 berupa variabel acak $h_* \sim \mathcal{N}(\mu_{h*}, \sigma_{h*}^2)$, integrasi eksak melalui fungsi kernel non-linier $k^{(2)}(h_*, h_i)$ bersifat intraktabel (\textit{intractable}). Salimbeni \& Deisenroth (2017) mengatasi kendala ini menggunakan \textbf{Trik Reparameterisasi Monte Carlo}:
\begin{equation}
    h_*^{(s)} = \mu_{h*} + \sigma_{h*} \cdot \epsilon^{(s)}, \quad \epsilon^{(s)} \sim \mathcal{N}(0, 1), \quad s = 1, \dots, S
\end{equation}

\subsection{Hukum Variansi Total (Law of Total Variance)}
\label{subsec:law_total_var}
Variansi akhir DGP diurai secara elegan menjadi dua komponen:
\begin{equation}
    \mathrm{Var}(y_*) = \underbrace{\mathbb{E}_{h_*}\left[ \mathrm{Var}(y_* \mid h_*) \right]}_{\text{Ketidakpastian Lokal Lapisan 2}} + \underbrace{\mathrm{Var}_{h_*}\left( \mathbb{E}[y_* \mid h_*] \right)}_{\text{Ketidakpastian Merambat dari Lapisan 1!}}
\end{equation}

\section{Contoh Perhitungan Numerik Lengkap (\texorpdfstring{$N = 3, N_* = 1$}{N=3, N*=1})}
\label{sec:perhitungan_numerik}

\begin{mathbox}
\textbf{Konfigurasi Masalah}:
\begin{itemize}
    \item 3 Titik Data Latih: $(x_1, y_1) = (1.0, 2.0)$, $(x_2, y_2) = (2.0, 1.0)$, $(x_3, y_3) = (4.0, 3.0)$.
    \item Representasi Laten Latih: $\mathbf{h} = [h_1, h_2, h_3]^T = [0.5, \; 1.2, \; 2.8]^T$.
    \item Titik Uji Celah Interpolasi: $x_* = 3.0$ (tidak memiliki data observasi).
    \item Parameter Kernel: $\sigma_{f1}^2 = \sigma_{f2}^2 = 1.0$, $l_1 = l_2 = 1.0$, $\sigma_{n1}^2 = \sigma_{n2}^2 = 0.01$.
\end{itemize}
\end{mathbox}

\subsection{Langkah 1: Pelatihan Lapisan 1 (\texorpdfstring{$X \to \mathbf{h}$}{X -> h})}
Matriks kovariansi masukan $K_y^{(1)} = K(X, X) + \sigma_{n1}^2 I$ dan inversnya:
\begin{equation}
K_y^{(1)} = \begin{bmatrix} 1.0100 & 0.6065 & 0.0111 \\ 0.6065 & 1.0100 & 0.1353 \\ 0.0111 & 0.1353 & 1.0100 \end{bmatrix}, \quad
(K_y^{(1)})^{-1} = \begin{bmatrix} 1.5606 & -0.9519 & 0.1104 \\ -0.9519 & 1.5889 & -0.2024 \\ 0.1104 & -0.2024 & 1.0160 \end{bmatrix}
\end{equation}

Vektor bobot dual Lapisan 1:
\begin{equation}
\boldsymbol{\alpha}^{(1)} = (K_y^{(1)})^{-1} \mathbf{h} = (K_y^{(1)})^{-1} \begin{bmatrix} 0.5 \\ 1.2 \\ 2.8 \end{bmatrix} = \begin{bmatrix} \mathbf{-0.0530} \\ \mathbf{0.8639} \\ \mathbf{2.6571} \end{bmatrix}
\end{equation}

\subsection{Langkah 2: Pelatihan Lapisan 2 (\texorpdfstring{$\mathbf{h} \to \mathbf{y}$}{h -> y})}
Jarak kernel Lapisan 2 dihitung antar fitur laten $\mathbf{h} = [0.5, 1.2, 2.8]^T$:
\begin{itemize}
    \item $k^{(2)}(h_1, h_2) = \exp\left(-0.5 \times |0.5 - 1.2|^2\right) = \exp(-0.245) = 0.7827$
    \item $k^{(2)}(h_1, h_3) = \exp\left(-0.5 \times |0.5 - 2.8|^2\right) = \exp(-2.645) = 0.0710$
    \item $k^{(2)}(h_2, h_3) = \exp\left(-0.5 \times |1.2 - 2.8|^2\right) = \exp(-1.280) = 0.2780$
\end{itemize}

Matriks kovariansi Lapisan 2 dan bobot dual $\boldsymbol{\alpha}^{(2)}$:
\begin{equation}
K_y^{(2)} = \begin{bmatrix} 1.0100 & 0.7827 & 0.0710 \\ 0.7827 & 1.0100 & 0.2780 \\ 0.0710 & 0.2780 & 1.0100 \end{bmatrix}, \quad
\boldsymbol{\alpha}^{(2)} = (K_y^{(2)})^{-1} \mathbf{y} = \begin{bmatrix} \mathbf{4.3212} \\ \mathbf{-3.3462} \\ \mathbf{3.5877} \end{bmatrix}
\end{equation}

\subsection{Langkah 3: Evaluasi Titik Interpolasi Baru (\texorpdfstring{$x_* = 3.0$}{x* = 3.0})}

\paragraph{A. Propagasi Maju Lapisan 1 ($x_* \to h_*$):}
Vektor kemiripan titik uji ke masukan latih: $\mathbf{k}_*^{(1)} = [0.1353, \; 0.6065, \; 0.6065]^T$.
\begin{align}
\mu_{h*} &= \mathbf{k}_*^{(1)T} \boldsymbol{\alpha}^{(1)} = (0.1353 \times -0.0530) + (0.6065 \times 0.8639) + (0.6065 \times 2.6571) = \mathbf{2.1284} \\
\sigma_{h*}^2 &= 1.0000 - \mathbf{k}_*^{(1)T} (K_y^{(1)})^{-1} \mathbf{k}_*^{(1)} = 1.0000 - 0.6998 = \mathbf{0.3002} \implies \sigma_{h*} = \mathbf{0.5479}
\end{align}
Luaran Lapisan 1 adalah distribusi: $h_* \sim \mathcal{N}\left(2.1284, \; 0.3002\right)$.

\paragraph{B. Reparameterization Sampling:}
Ditarik 3 sampel representatif $h_*^{(s)} = \mu_{h*} + \sigma_{h*} \epsilon^{(s)}$:
\begin{itemize}
    \item \textbf{Sampel 1} ($\epsilon^{(1)} = 0$): $h_*^{(1)} = 2.1284$ (Nilai mean)
    \item \textbf{Sampel 2} ($\epsilon^{(2)} = +1$): $h_*^{(2)} = 2.1284 + 0.5479 = \mathbf{2.6763}$ (Mendekati $h_3 = 2.8$)
    \item \textbf{Sampel 3} ($\epsilon^{(3)} = -1$): $h_*^{(3)} = 2.1284 - 0.5479 = \mathbf{1.5805}$ (Mendekati $h_2 = 1.2$)
\end{itemize}

\paragraph{C. Propagasi Maju Lapisan 2 untuk Masing-masing Sampel ($h_*^{(s)} \to y_*^{(s)}$):}
\begin{enumerate}
    \item \textbf{Sampel 1 ($h_*^{(1)} = 2.1284$)}: \\
    $\mathbf{k}_*^{(2)(1)} = [0.2656, \; 0.6499, \; 0.7981]^T$ \\
    $\mu_{y*}^{(1)} = \mathbf{k}_*^{(2)(1)T} \boldsymbol{\alpha}^{(2)} = \mathbf{1.8363}, \quad \sigma_{y*}^{2(1)} = 1.0 - \mathbf{k}_*^{(2)(1)T} (K_y^{(2)})^{-1} \mathbf{k}_*^{(2)(1)} = \mathbf{0.1180}$
    
    \item \textbf{Sampel 2 ($h_*^{(2)} = 2.6763$)}: \\
    $\mathbf{k}_*^{(2)(2)} = [0.0936, \; 0.3363, \; \mathbf{0.9924}]^T$ (sangat mirip dengan $h_3 = 2.8$) \\
    $\mu_{y*}^{(2)} = \mathbf{2.8397}$ (naik mendekati target $y_3 = 3.0$), $\quad \sigma_{y*}^{2(2)} = \mathbf{0.0186}$ (variansi sangat kecil).
    
    \item \textbf{Sampel 3 ($h_*^{(3)} = 1.5805$)}: \\
    $\mathbf{k}_*^{(2)(3)} = [0.5578, \; \mathbf{0.9302}, \; 0.4754]^T$ (sangat mirip dengan $h_2 = 1.2$) \\
    $\mu_{y*}^{(3)} = \mathbf{1.0034}$ (turun mendekati target $y_2 = 1.0$), $\quad \sigma_{y*}^{2(3)} = \mathbf{0.0481}$.
\end{enumerate}

\paragraph{D. Agregasi Luaran via Hukum Variansi Total:}
\begin{align}
\mu_{\mathrm{final}} &= \frac{1}{3}\left( 1.8363 + 2.8397 + 1.0034 \right) = \mathbf{1.8931} \\
\mathbb{E}[\sigma_y^2] &= \frac{0.1180 + 0.0186 + 0.0481}{3} = \mathbf{0.0616} \quad (\text{Ketidakpastian lokal L2}) \\
\mathrm{Var}(\mu_y) &= \frac{1}{3}\sum_{s=1}^3 (\mu_{y*}^{(s)} - \mu_{\mathrm{final}})^2 = \mathbf{0.5636} \quad (\text{Perambatan dari ketidakpastian laten L1}) \\
\sigma^2_{\mathrm{total}} &= 0.0616 + 0.5636 = \mathbf{0.6252} \implies \sigma_{\mathrm{total}} = \sqrt{0.6252} = \mathbf{0.7907}
\end{align}
